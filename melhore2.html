// Criação da função para evitar repetição function exibirInformacoes(nome, idade, curso) { console.log("-------------------------------------"); console.log("Nome: " + nome); console.log("Idade: " + idade); console.log("Curso: " + curso); console.log("-------------------------------------"); } // Dados do primeiro aluno let nome = "Carlos"; let idade = 17; let curso = "Desenvolvimento de Sistemas"; // Chamada da função para o Carlos exibirInformacoes(nome, idade, curso); // Dados do segundo aluno let nome2 = "Ana"; let idade2 = 16; let curso2 = "Desenvolvimento de Sistemas"; // Chamada da função para a Ana exibirInformacoes(nome2, idade2, curso2);
